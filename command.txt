All systems command
You are extending the SQLAlchemy models for a football chairman simulation.
All models go in engine/models/. Do not touch engine/models/base.py unless 
adding a relationship. Create new files for new model groups.

The existing models are:
  base.py      → Club, Player, League, Fixture, Stadium, GameMeta
  history.py   → ClubHistory, NewsEvent
  personality.py → Personality

TASK 1 — Create engine/models/contracts.py:
  PlayerContract:
    id, player_id (FK players), club_id (FK clubs)
    wage (Integer), start_date (DateTime), end_date (DateTime)
    release_clause (Integer, nullable)
    status: 'ACTIVE' | 'EXPIRED' | 'TERMINATED'
    
  ManagerContract:
    id, manager_id (FK managers), club_id (FK clubs)
    wage (Integer), start_date (DateTime), end_date (DateTime)
    sack_compensation (Integer)
    status: 'ACTIVE' | 'EXPIRED' | 'TERMINATED'

TASK 2 — Create engine/models/staff.py:
  Manager:
    id, name, age, nationality
    tactical_style: JSON  (e.g. {"preferred_formation": "4-3-3", "style": "pressing"})
    reputation (Integer 0-10000)
    preferred_formations: JSON (list of strings)
    man_management (Integer 1-20)
    tactical_knowledge (Integer 1-20)
    youth_development (Integer 1-20)
    ambition (Integer 1-20)
    club_id (FK clubs, nullable — may be unemployed)
    morale (Integer 0-100)
    demands: JSON  (e.g. {"transfer_budget": 5000000, "promises": ["sign_striker"]})

  Scout:
    id, name, club_id (FK clubs)
    ability (Integer 1-20)
    judgment (Integer 1-20)
    network: JSON  (list of country strings they cover well)

TASK 3 — Create engine/models/transfers.py:
  TransferOffer:
    id, player_id (FK players)
    from_club_id (FK clubs), to_club_id (FK clubs)
    fee (Integer), status: 'PENDING' | 'ACCEPTED' | 'REJECTED' | 'WITHDRAWN'
    created_date (DateTime), resolved_date (DateTime, nullable)
    installments (Integer, default 1)
    is_loan (Boolean, default False)

  ScoutReport:
    id, player_id (FK players), scout_id (FK scouts)
    club_id (FK clubs)
    created_date (DateTime)
    ca_estimate (Integer)
    pa_estimate (Integer)
    notes (String)
    attributes_snapshot: JSON

TASK 4 — Create engine/models/finance.py:
  FinancialRecord:
    id, club_id (FK clubs), date (DateTime)
    record_type: 'WAGE' | 'TRANSFER_IN' | 'TRANSFER_OUT' | 
                 'TICKET' | 'SPONSORSHIP' | 'TV_RIGHTS' | 'OTHER'
    amount (Integer)  — positive = income, negative = expense
    description (String)

  Sponsorship:
    id, club_id (FK clubs)
    sponsor_name (String)
    amount_per_season (Integer)
    start_season (Integer), end_season (Integer)
    status: 'ACTIVE' | 'EXPIRED'

TASK 5 — Create engine/models/world.py:
  Season:
    id, league_id (FK leagues)
    season_number (Integer)  — e.g. 1, 2, 3 representing years
    start_date (DateTime), end_date (DateTime)
    status: 'ACTIVE' | 'COMPLETED'
    champion_club_id (FK clubs, nullable)

  LeagueStanding:
    id, season_id (FK seasons), club_id (FK clubs)
    played, won, drawn, lost, gf, ga, gd, points (all Integer)
    form: JSON  (last 5 results as list of 'W'/'D'/'L')

  Rivalry:
    id, club_a_id (FK clubs), club_b_id (FK clubs)
    intensity (Integer 1-10)
    rivalry_type: 'LOCAL' | 'HISTORIC' | 'TITLE'

  Facilities:

    youth_upgrade_cost (Integer)
